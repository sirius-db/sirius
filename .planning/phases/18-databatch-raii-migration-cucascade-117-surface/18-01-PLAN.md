namespace cucascade {

class data_batch : public std::enable_shared_from_this<data_batch> {
public:
  uint64_t get_batch_id() const;
  batch_state get_state() const;                    // {idle, read_only, mutable_locked}
  size_t get_read_only_count() const;
  size_t get_subscriber_count() const;
  void subscribe();
  void unsubscribe();

  [[nodiscard]] read_only_data_batch to_read_only();             // BLOCKS
  [[nodiscard]] mutable_data_batch  to_mutable();                // BLOCKS
  [[nodiscard]] std::optional<read_only_data_batch> try_to_read_only();
  [[nodiscard]] std::optional<mutable_data_batch>   try_to_mutable();

  [[nodiscard]] static std::shared_ptr<data_batch> to_idle(read_only_data_batch&&);
  [[nodiscard]] static std::shared_ptr<data_batch> to_idle(mutable_data_batch&&);
  [[nodiscard]] static mutable_data_batch readonly_to_mutable(read_only_data_batch&&);
  [[nodiscard]] static read_only_data_batch mutable_to_readonly(mutable_data_batch&&);
};

class read_only_data_batch {
public:
  uint64_t get_batch_id() const;
  memory::Tier get_current_tier() const;
  idata_representation* get_data() const;            // PUBLIC HERE
  memory::memory_space* get_memory_space() const;    // PUBLIC HERE
  [[nodiscard]] cudaEvent_t get_writer_event() const;
};

class mutable_data_batch {
public:
  // same readers as read_only_data_batch

                  const memory::memory_space* target_space,
                  rmm::cuda_stream_view stream);     // synchronizes stream internally
};

}  // namespace cucascade
