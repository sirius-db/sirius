    ```cpp
    // test/cpp/integration/test_gpu_execution_tpch_mgpu_audit.cpp
    // AUDIT-01/02/03 (v1.2): runs a multi-batch TPC-H query on num_gpus=2 and
    // asserts per-GPU task/batch distribution via [mgpu-audit] log grep.
    //
    // Design (per 08-RESEARCH.md Pitfall 3):
    //   1. pause g_integration_env_2gpu so SiriusContext is not yet initialized
    //   2. setenv SIRIUS_LOG_DIR to a unique tmp path
    //   3. resume env — fresh SiriusContext picks up the new log dir
    //   4. run query via compare_gpu_vs_cpu (disables fallback)
    //   5. pause env (forces log flush)
    //   6. grep the log file for [mgpu-audit] pipeline_task + scan_batch per GPU
    //   7. count UNIQUE task_id / batch_id per GPU via sort -u (or equivalent in-process)
    //   8. REQUIRE counts >= 5 per GPU per event type
    //
    // Single-GPU hosts: cudaGetDeviceCount() < 2 triggers WARN+return (per Catch2 v2 convention).

    #include <catch2/catch.hpp>
    #include <cuda_runtime.h>
    #include <cstdlib>
    #include <filesystem>
    #include <fstream>
    #include <regex>
    #include <set>
    #include <string>
    #include <duckdb.hpp>

    #include "utils/sirius_test_env.hpp"

    namespace fs = std::filesystem;

    namespace {
    struct AuditCounts { std::set<std::string> pipeline_ids; std::set<std::string> scan_ids; };

    std::map<int, AuditCounts> parse_audit_log(const fs::path& log_dir) {
      std::map<int, AuditCounts> by_gpu;
      // Match e.g.: "[mgpu-audit] pipeline_task dispatched to GPU 0 task_id=139813xx"
      std::regex pipeline_re(R"(\[mgpu-audit\] pipeline_task dispatched to GPU (\d+) task_id=(\S+))");
      std::regex scan_re(R"(\[mgpu-audit\] scan_batch assigned to GPU (\d+) batch_id=(\S+))");
      for (auto const& entry : fs::directory_iterator(log_dir)) {
        if (!entry.is_regular_file()) continue;
        std::ifstream f(entry.path());
        std::string line;
        while (std::getline(f, line)) {
          std::smatch m;
          if (std::regex_search(line, m, pipeline_re)) {
            by_gpu[std::stoi(m[1])].pipeline_ids.insert(m[2]);
          } else if (std::regex_search(line, m, scan_re)) {
            by_gpu[std::stoi(m[1])].scan_ids.insert(m[2]);
          }
        }
      }
      return by_gpu;
    }
    }  // namespace

    TEST_CASE("tpch_q1 multi-gpu audit distribution", "[tpch][mgpu-audit]") {
      int device_count = 0;
      cudaGetDeviceCount(&device_count);
      if (device_count < 2) {
        WARN("AUDIT test requires >= 2 GPUs; single-GPU host — skipping (per AUDIT-03 Catch2-v2 WARN+return)");
        return;
      }

      auto* env = sirius::test::g_integration_env_2gpu;
      REQUIRE(env != nullptr);

      auto tmp_log_dir = fs::temp_directory_path() /
                         ("sirius-mgpu-audit-" + std::to_string(::getpid()));
      fs::remove_all(tmp_log_dir);
      fs::create_directories(tmp_log_dir);

      env->pause();
      setenv("SIRIUS_LOG_DIR", tmp_log_dir.c_str(), 1);
      setenv("SIRIUS_LOG_LEVEL", "info", 1);
      env->resume();

      {
        auto con = std::make_unique<duckdb::Connection>(env->make_connection());
        con->Query("SET enable_duckdb_fallback = false;");
        // TPC-H Q1 on the bundled SF1 integration dataset — produces multiple scan batches
        // with scan_task_batch_size at whatever integration-2gpu.yaml configures.
        // ROADMAP criterion 4 requires >= 5 on BOTH GPUs; if SF1 is too small, scale up
        // via scan_task_batch_size override in integration-2gpu.yaml OR rely on SF10 variant.
        constexpr auto kQ1 = R"sql(
          CALL gpu_execution('
            SELECT l_returnflag, l_linestatus,
                   sum(l_quantity), sum(l_extendedprice), avg(l_discount)
              FROM lineitem
             WHERE l_shipdate <= DATE '1998-12-01' - INTERVAL 90 DAY
             GROUP BY l_returnflag, l_linestatus
             ORDER BY l_returnflag, l_linestatus
          ')
        )sql";
        auto result = con->Query(kQ1);
        INFO("Q1 result error (if any): " << (result->HasError() ? result->GetError() : std::string{}));
        REQUIRE(!result->HasError());
      }

      env->pause();  // flush logs, release connection

      // Parse the log; count unique task_ids and batch_ids per GPU
      auto counts = parse_audit_log(tmp_log_dir);

      // Build a diagnostic message for failure cases
      std::string diag;
      for (auto const& [gpu, c] : counts) {
        diag += "GPU " + std::to_string(gpu) +
                ": pipeline=" + std::to_string(c.pipeline_ids.size()) +
                " scan=" + std::to_string(c.scan_ids.size()) + "; ";
      }
      INFO("Per-GPU audit counts: " << diag);

      // ROADMAP criterion 4: >= 5 pipeline_task AND >= 5 scan_batch on BOTH GPUs
      REQUIRE(counts.count(0) == 1);
      REQUIRE(counts.count(1) == 1);
      REQUIRE(counts[0].pipeline_ids.size() >= 5);
      REQUIRE(counts[1].pipeline_ids.size() >= 5);
      REQUIRE(counts[0].scan_ids.size() >= 5);
      REQUIRE(counts[1].scan_ids.size() >= 5);

      // Cleanup
      unsetenv("SIRIUS_LOG_DIR");
      fs::remove_all(tmp_log_dir);
    }
    ```
