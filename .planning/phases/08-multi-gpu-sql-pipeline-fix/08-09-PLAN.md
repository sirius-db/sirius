```cpp
  // FIX-02 gap-closure (08-09 hypothesis A): pin the thread to the target
  // device for the duration of compute_task. The upstream H2D read chain
  // (read_range_into_allocation -> prefetched_data_source -> cudaMemcpyAsync)
  // uses the CALLER's stream, which under num_gpus=2 may be bound to a
  // different device than the memspace we're writing to. The Pattern 2
  // idiom (src/data/sirius_host_to_gpu_converter.cpp:255) requires the
  // current thread device == target device before any H2D copy lands; this
  // RAII enforces that for the pre-converter frame.
  auto const& _io_backends = g_state.get_gpu_io_backends();
  int const target_device_probe =
    g_state.get_preferred_device_id().has_value()
      ? *g_state.get_preferred_device_id()
      : (_io_backends.empty() ? 0 : _io_backends.begin()->first);
  rmm::cuda_set_device_raii compute_task_device_raii{
    rmm::cuda_device_id{target_device_probe}};
```
