class convertible_data_batch : public convertible_data {

 public:
  explicit convertible_data_batch_provider(cucascade::shared_data_repository* repo);

    cucascade::memory::memory_space* space, bool front_to_back) override;
  std::vector<std::unique_ptr<convertible_data>> get_all_convertible(
    cucascade::memory::memory_space* space, bool front_to_back) override;
  std::size_t get_bytes_in_space(cucascade::memory::memory_space* space) const override;
};

}  // namespace sirius

**Test cases (all tagged `[convertible_data_batch]`):**

**TEST_CASE 1: "convertible_data_batch converts GPU batch to HOST"**
- Create batch: `auto batch = sirius::test::operator_utils::make_numeric_batch(*env().gpu_space, std::vector<int32_t>{1, 2, 3, 4, 5}, cudf::type_id::INT32);`
- Verify initial state: `REQUIRE(batch->get_memory_space() == env().gpu_space);` and `REQUIRE(batch->get_state() == cucascade::batch_state::idle);`
- Create wrapper: `sirius::convertible_data_batch wrapper(batch);`
- Call convert: `bool result = wrapper.convert({env().host_space}, env().stream, *env().mgr);`
- `REQUIRE(result == true);`
- `REQUIRE(batch->get_memory_space()->get_tier() == cucascade::memory::Tier::HOST);`
- `REQUIRE(batch->get_state() == cucascade::batch_state::idle);`

**TEST_CASE 2: "convertible_data_batch returns false with empty target_spaces"**
- Create a GPU batch.
- `bool result = wrapper.convert({}, env().stream, *env().mgr);`
- `REQUIRE(result == false);`
- Batch memory space still GPU, state still idle.

**TEST_CASE 3: "convertible_data_batch_provider get_next_convertible returns last idle batch"**
- Create `cucascade::shared_data_repository repo;`
- Add 3 batches via `repo.add_data_batch(batch)`. Make the third one `task_created` state by calling `batch3->try_to_create_task()`.
- Create provider: `sirius::convertible_data_batch_provider provider(&repo);`
- Call `auto cd = provider.get_next_convertible(env().gpu_space, false);`
- `REQUIRE(cd != nullptr);`
- Verify it returns batch2 (the last idle batch when iterating last-to-first) by checking `cd->bytes_in_space(env().gpu_space)` matches batch2's size.

**TEST_CASE 4: "convertible_data_batch_provider get_all_convertible returns all idle batches"**
- Same setup as Test 3.
- `auto all = provider.get_all_convertible(env().gpu_space, false);`
- `REQUIRE(all.size() == 2);` (batch1 and batch2 are idle; batch3 is task_created)

**TEST_CASE 5: "convertible_data_batch_provider iterates multi-partition last-to-first"**
- Create `cucascade::shared_data_repository repo;`
- Add batch_p0 to partition 0: `repo.add_data_batch(batch_p0, 0);`
- Add batch_p1 to partition 1: `repo.add_data_batch(batch_p1, 1);`
- Provider with `front_to_back=false` should find batch_p1 first (partition 1 before partition 0).
- `auto cd = provider.get_next_convertible(env().gpu_space, false);`
- Verify the returned batch is from partition 1 by comparing bytes_in_space.

**TEST_CASE 6: "convertible_data_batch convert fails when batch already in_transit"**
- Create GPU batch, manually lock: `batch->try_to_lock_for_in_transit();`
- `REQUIRE(batch->get_state() == cucascade::batch_state::in_transit);`
- `sirius::convertible_data_batch wrapper(batch);`
- `bool result = wrapper.convert({env().host_space}, env().stream, *env().mgr);`
- `REQUIRE(result == false);`
- `REQUIRE(batch->get_state() == cucascade::batch_state::in_transit);` (state unchanged, per BATCH-03)
- Release manually: `batch->try_to_release_in_transit();`

**TEST_CASE 7: "convertible_data_batch bytes_in_space returns correct size"**
- Create GPU batch.
- `sirius::convertible_data_batch wrapper(batch);`
- `auto size = wrapper.bytes_in_space(env().gpu_space);`
- `REQUIRE(size > 0);`
- `REQUIRE(size == batch->get_data()->get_size_in_bytes());`
- `REQUIRE(wrapper.bytes_in_space(env().host_space) == 0);`

**TEST_CASE 8: "convertible_data_batch_provider get_bytes_in_space sums batch sizes"**
- Create repo with 2 GPU batches.
- `sirius::convertible_data_batch_provider provider(&repo);`
- `auto total = provider.get_bytes_in_space(env().gpu_space);`
- `REQUIRE(total == batch1_size + batch2_size);`
- `REQUIRE(provider.get_bytes_in_space(env().host_space) == 0);`
