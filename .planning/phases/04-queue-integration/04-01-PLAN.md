```cpp
namespace sirius::exec {

template <typename T>
class inspectable_mpsc {
 public:
  inspectable_mpsc()  = default;
  ~inspectable_mpsc() = default;

  // Deleted copy/move
  inspectable_mpsc(const inspectable_mpsc&)            = delete;
  inspectable_mpsc& operator=(const inspectable_mpsc&) = delete;
  inspectable_mpsc(inspectable_mpsc&&)                 = delete;
  inspectable_mpsc& operator=(inspectable_mpsc&&)      = delete;

  [[nodiscard]] bool push(std::unique_ptr<T> item);
  template <typename... Args>
  [[nodiscard]] bool emplace(Args&&... args);
  std::unique_ptr<T> pop();         // blocking, returns nullptr on interrupt
  std::unique_ptr<T> try_pop();     // non-blocking, returns nullptr if empty
  void interrupt();
  void reactivate();
  void drain();
  [[nodiscard]] bool is_open() const noexcept;
  [[nodiscard]] bool is_empty() const noexcept;
  [[nodiscard]] std::size_t size() const noexcept;

  // Predicate inspection (new capability):
  std::unique_ptr<T> pop_if(std::function<bool(const T&)> predicate, bool front_to_back);
  T* get_if(std::function<bool(const T&)> predicate, bool front_to_back);
  std::unique_ptr<T> mutable_pop_if(std::function<bool(T&)> predicate, bool front_to_back);
  T* mutable_get_if(std::function<bool(T&)> predicate, bool front_to_back);
};

}  // namespace sirius::exec
```

From src/include/exec/interruptible_mpmc.hpp (the queue being replaced):
```cpp
namespace sirius::exec {

template <smart_pointer T>
class interruptible_mpmc {
 public:
  bool push(pointer_type item);       // NOT [[nodiscard]]
  pointer_type pop();                 // blocking, returns nullptr on interrupt
  pointer_type try_pop();             // non-blocking
  void interrupt();
  void reactivate();
  void drain();
  [[nodiscard]] bool is_open() const noexcept;
  [[nodiscard]] bool is_empty() const noexcept;
};

}  // namespace sirius::exec
```

CRITICAL API DIFFERENCE:
- interruptible_mpmc<std::unique_ptr<itask>> is parameterized on the smart pointer itself
- inspectable_mpsc<itask> is parameterized on the element type (wraps in unique_ptr internally)
- So: `interruptible_mpmc<std::unique_ptr<itask>>` becomes `inspectable_mpsc<itask>`
- inspectable_mpsc::push() is [[nodiscard]] -- callers must handle the bool return
