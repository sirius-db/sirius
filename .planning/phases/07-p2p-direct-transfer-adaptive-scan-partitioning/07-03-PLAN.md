```cpp
int duckdb_scan_executor::select_target_gpu()
{
  if (_gpu_memory_spaces.size() <= 1) {
    return _gpu_memory_spaces.empty() ? 0 : _gpu_memory_spaces[0]->get_device_id();
  }
  size_t total_available = 0;
  for (auto* space : _gpu_memory_spaces) {
    total_available += space->get_available_memory();
  }
  if (total_available == 0) {
    auto idx = _scan_round_robin.fetch_add(1) % _gpu_memory_spaces.size();
    return _gpu_memory_spaces[idx]->get_device_id();
  }
  auto counter      = _scan_round_robin.fetch_add(1);
  size_t target     = counter % total_available;
  size_t cumulative = 0;
  for (auto* space : _gpu_memory_spaces) {
    cumulative += space->get_available_memory();
    if (target < cumulative) {
      SIRIUS_LOG_DEBUG("Scan executor: distributing scan batch to GPU {} (available: {} bytes)",
                       space->get_device_id(),
                       space->get_available_memory());
      return space->get_device_id();
    }
  }
  return _gpu_memory_spaces.back()->get_device_id();
}
```

   ```cpp
   std::vector<cucascade::memory::memory_space*> spaces = {
     const_cast<cucascade::memory::memory_space*>(gpu_spaces[0]),
     const_cast<cucascade::memory::memory_space*>(gpu_spaces[1])
   };
   size_t total_available = 0;
   for (auto* s : spaces) total_available += s->get_available_memory();

   std::atomic<uint64_t> counter{0};
   std::unordered_map<int, int> histogram;
   for (int i = 0; i < 32; ++i) {
     auto c = counter.fetch_add(1);
     size_t target = c % total_available;
     size_t cumulative = 0;
     for (auto* s : spaces) {
       cumulative += s->get_available_memory();
       if (target < cumulative) {
         histogram[s->get_device_id()]++;
         break;
       }
     }
   }
   REQUIRE(histogram.size() == 2);
   int count_gpu0 = histogram[gpu_spaces[0]->get_device_id()];
   int count_gpu1 = histogram[gpu_spaces[1]->get_device_id()];
   REQUIRE(count_gpu1 > count_gpu0);   // GPU 1 has more free -> more batches
   double batch_ratio = static_cast<double>(count_gpu1) / static_cast<double>(std::max(count_gpu0, 1));
   REQUIRE(batch_ratio >= 2.0);        // skew ≥ 2× (CONTEXT success criterion 3)
   double expected_ratio = free_ratio_gpu1_over_gpu0;
   double delta = std::abs(batch_ratio - expected_ratio) / expected_ratio;
   REQUIRE(delta <= 0.10);             // within 10% tolerance (CONTEXT lock)
   ```
