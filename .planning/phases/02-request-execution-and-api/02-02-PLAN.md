The `shared_data_repository_manager` does NOT have an `add_repository(T*)` method. The correct API is:

9. **`"request_downgrade with custom predicate stops when satisfied"` (RAPI-01, RAPI-04, EXEC-04)**:
```cpp
TEST_CASE("request_downgrade with custom predicate stops when satisfied",
          "[downgrade_executor]")
{
  auto mem_mgr    = make_test_memory_manager();
  auto* gpu_space = get_gpu_space(*mem_mgr);
  REQUIRE(gpu_space != nullptr);

  cucascade::shared_data_repository_manager repo_mgr;
  auto repo = std::make_unique<cucascade::shared_data_repository>();
  std::vector<std::shared_ptr<cucascade::data_batch>> batches;
  for (int i = 0; i < 5; ++i) {
    auto batch = make_gpu_batch(*gpu_space);
    batches.push_back(batch);
    repo->add_data_batch(batch);
  }
  repo_mgr.add_new_repository(1, "out", std::move(repo));

  std::atomic<size_t> external_counter{0};
  size_t one_batch_size = batches[0]->get_data()->get_size_in_bytes();

  auto executor = make_test_executor(repo_mgr, gpu_space, *mem_mgr);
  executor.start();

  // Predicate: stop after first batch worth of data freed
  // The predicate doesn't have access to bytes_freed, so use an external counter.
  // Actually, the predicate can be anything. For this test, use a simple counter:
  auto future = executor.request_downgrade([&external_counter, one_batch_size]() {
    // Note: the predicate is called from the dispatch lambda AFTER bytes_freed
    // is incremented. But the predicate doesn't see bytes_freed directly;
    // it's a user-provided predicate. For testing, we increment our own counter.
    external_counter.fetch_add(1, std::memory_order_relaxed);
    return external_counter.load(std::memory_order_relaxed) >= 1;
  });

  size_t freed = future.get();
  REQUIRE(freed > 0);

  // Should have stopped after ~1 batch (plus any in-flight)
  size_t host_count = 0;
  for (auto& b : batches) {
    if (b->get_memory_space()->get_tier() == cucascade::memory::Tier::HOST) ++host_count;
  }
  REQUIRE(host_count >= 1);
  // With pool width = 1, should stop after 1-2 batches
  REQUIRE(host_count <= 2);

  executor.stop();
}
```
