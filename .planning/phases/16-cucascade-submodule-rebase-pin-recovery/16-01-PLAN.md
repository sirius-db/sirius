# Use GIT_SEQUENCE_EDITOR to script the rebase todo-list,
# and GIT_EDITOR to accept the auto-generated squash message body
# (we'll overwrite it after each squash via git commit --amend -F).

export GIT_SEQUENCE_EDITOR="bash -c 'cat > \$1 <<EOF
pick 1fff85d
squash 3743621
squash 2dcab24
squash ff14ff4
squash e23f3a2
pick 7ed84f2
squash cc2a53d
squash e4db3d8
pick eda349a
pick 7409c60
squash 62e0517
EOF' --"

# GIT_EDITOR=true accepts whatever message git generated. We rewrite each
# of the 4 group messages immediately after rebase completes, in commit order.
export GIT_EDITOR=true

git rebase -i edd6f03

# After the squash rebase completes, HEAD is 4 commits above edd6f03.
# Now rewrite each of the 4 group commit messages, oldest -> newest, using
# a targeted reword-the-Nth-pick rebase scripted via GIT_SEQUENCE_EDITOR.
# Each iteration rewrites one group message; after 4 iterations all messages
# are in their final form.

reword_nth() {
  local n=$1            # 1=oldest, 4=newest
  local msg_file=$2
  GIT_SEQUENCE_EDITOR="bash -c 'awk -v N=$n \"BEGIN{c=0} /^pick / {c++; if (c==N) {sub(/^pick/,\\\"reword\\\")}} {print}\" \$1 > \$1.new && mv \$1.new \$1' --" \
  GIT_EDITOR="bash -c 'cat $msg_file > \$1' --" \
  git rebase -i HEAD~4
}

reword_nth 1 "$TMPDIR/phase16-msgs/group1.txt"
reword_nth 2 "$TMPDIR/phase16-msgs/group2.txt"
reword_nth 3 "$TMPDIR/phase16-msgs/group3.txt"
reword_nth 4 "$TMPDIR/phase16-msgs/group4.txt"

# Final verification:
git log --oneline edd6f03..HEAD
